
Table users { //사용자
  id varchar(100) [primary key]              //ID
  password varchar(255) [not null]           //암호화된 비밀번호(Hash) -> JWT
  email varchar(100) [not null]              //이메일(찾기시 사용)
  role varchar(20)                           //권한(회원 null, 관리자)
  nickname varchar(30) [not null]            //게시판에서 사용할 닉네임
  score integer [default: 0]                 //활동 점수(게시판 글, 리뷰 작성 등)
  grade varchar(20) [default: '방린이']       //등급
  created_at timestamp [not null, default: `now()`] //가입일
  updated_at timestamp                       //수정일

  indexes {//UNIQUE 제약조건 (중복 방지: uk_테이블명_컬럼명)
    email [unique, name: 'uk_users_email']
    nickname [unique, name: 'uk_users_nickname']
  }
}

Table posts { //게시글
  id integer [pk, increment]                 //게시글 번호
  user_id varchar(100) [not null]            //작성자
  category varchar(50) [not null]            //카테고리
  title varchar(200) [not null]              //제목
  content text [not null]                    //내용
  view_count integer [not null, default: 0]  //조회수
  //accepted_id integer                      //채택된 댓글 -> 댓글에서 채택여부 1인 값이 있는지 체크 해서 기능으로 관리
  is_spoiler tinyint [not null, default: 0]  //스포일러 여부(스포일러 1)
  is_notice tinyint [not null, default: 0]   //공지사항 여부(공지사항 1) 관리자만
  created_at timestamp [not null, default: `now()`] //작성일
  updated_at timestamp                       //수정일
  
  /* indexes {//UNIQUE 제약조건 (중복 방지: uk_테이블명_컬럼명)
    accepted_id [unique, name: 'uk_posts_accepted_id']
  } */
  indexes {
    created_at [name: 'idx_posts_created_at']
  }
}

Table posts_logs { //게시글 수정 로그
  id integer [pk, increment]                        //로그 번호(PK)
  post_id integer [not null]                        //대상 게시글 번호
  user_id varchar(100) [not null]                   //수정한 유저/관리자 ID
  category varchar(50) [not null]                   //수정 당시 카테고리
  title varchar(200) [not null]                     //수정 당시 제목
  content text [not null]                           //수정 당시 내용
  is_spoiler tinyint [not null]                     //수정 당시 스포일러 여부
  is_notice tinyint [not null]                      //수정 당시 공지사항 여부
  created_at timestamp [not null, default: `now()`] //로그 기록일(수정 일시)
}

Table post_reactions { //좋아요(수정 없음: 등록/취소(삭제)만 존재)
  id integer [pk, increment]          //좋아요 //이부분에 대해서 설명 확인 필요
  post_id integer [not null]          //게시글 번호(FK)
  user_id varchar(100) [not null]     //작성자(FK)
  //reaction_type varchar             //좋아요/싫어요/null
  //reaction_type tinyint             //좋아요(1)/취소(0)
  created_at timestamp [not null, default: `now()`] //작성일

  indexes {//UNIQUE 제약조건 (중복 방지: uk_테이블명_컬럼명)

  }
}

Table comments { //댓글 (수정 불가)

  is_accepted tinyint [not null, default: 0]        //채택 여부(기본 채택안함 0, 채택1)

    created_at [name: 'idx_comments_created_at']
  }
}

Table themes { //테마(관리자)
  id integer [pk, increment]                        //테마 번호(PK)
  user_id varchar(100) [not null]                   //작성자
  title varchar(150) [not null]                     //테마명
  store_name varchar(100) [not null]                //매장명(예: 강남점, 홍대본점)
  address varchar(255) [not null]                   //도로명/지번 주소
  //map_url varchar(500)                            //지도 좌표 정보 -> X와 Y로 각각 저장하기로 하여 제거
  latitude decimal(10,7)                            //지도 Y 좌표 정보 
  longitude decimal(10,7)                           //지도 X 좌표 정보 
  //image_url varchar(500)                          //해당 테마의 이미지 URL (회의시 우선 빼고 추후 고려로 확인)
  thumbnail_url varchar(2048)                       //해당 테마의 이미지(썸네일) URL -> 필드명 변경 및 외부 이미지 캐싱 URL 안정성을 위해 길이 2048로 확인 
  content text [not null]                           //테마 스토리 및 소개 내용
  min_people tinyint [not null, default: 0]         //최소 인원수
  max_people tinyint [not null, default: 0]         //최대 인원수
  play_time integer [not null, default: 60]         //플레이타임(분 단위)
  difficulty tinyint [not null]                     //난이도 (1 ~ 5)
  price integer [not null, default: 0]              //1인당 기본 가격
  genre varchar(50) [not null]                      //장르
  created_at timestamp [not null, default: `now()`] //생성일
  updated_at timestamp                              //수정일
  
  indexes {
    created_at [name: 'idx_themes_created_at']
  }
}

Table theme_logs { //테마 수정 로그

  //map_url varchar(500)                            //수정 당시 지도 정보

  max_people tinyint [not null]                     //수정 당시 최대 인원수

Table theme_reviews { //후기/평점 (수정 불가)
  id integer [pk, increment]                        //후기 번호(PK)
  theme_id integer [not null]                       //대상 테마 번호(FK)
  user_id varchar(100) [not null]                   //작성자(FK)
  content text [not null]                           //후기 내용
  rating tinyint [not null]                         //평점(1 ~ 5)
  created_at timestamp [not null, default: `now()`] //작성일

  indexes {//UNIQUE 제약조건 (중복 방지: uk_테이블명_컬럼명)

    created_at [name: 'idx_theme_reviews_created_at']

Table recruitments { //모집글 테이블
  id int [pk, increment]                            //모집글 번호 (PK)
  user_id varchar(100) [not null]                   //작성자 ID (FK)
  theme_id int [null]                               //방탈출 테마 ID (FK) -> 테마명 직접 입력시 연결하지 않으므로 null
  theme_name varchar(150) [null]                    //방탈출 테마명(직접 입력 가능)
  title varchar(200) [not null]                     //모집글 제목
  content text [not null]                           //모집글 내용
  meet_date datetime [not null]                     //모임 일시
  capacity int [not null]                           //모집 정원
  status varchar(20) [not null, default: 'OPEN']    //상태 (OPEN(모집중), CLOSED 마감(인원충족), CANCELED (모집취소,인원미달), APPROVED(모집 완료))
  created_at timestamp [not null, default: `now()`] //생성일시
  updated_at timestamp [null]                       //수정일시(수정이 상태값 바꿀때 사용되므로 필드는 유지하고 상태값만 바꾸고 따로 수정은 안되도록 구현, 로그 테이블 및 관리는 기간상 미진행)
}

Table recruitment_applications { //모집 신청(댓글) 테이블
  id int [pk, increment]                              //신청 번호 (PK)
  recruitment_id int [not null]                       //모집글 ID (FK)
  user_id varchar(100) [not null]                     //신청자 ID (FK)
  content text [not null]                             //신청 한마디
  status varchar(20) [not null, default: 'PENDING']   //상태 (PENDING (대기 중), ACCEPTED (수락됨/참가 확정), REJECTED (거절됨(선택 안된 경우)))
  created_at timestamp [not null, default: `now()`]   //신청일시

  indexes {
    (recruitment_id, user_id) [unique, name: 'uq_recruitment_user']
  }
}

Table party_comments { //파티 댓글 테이블
  id int [pk, increment]                            //파티 댓글 번호 (PK)
  recruitment_id int [not null]                     //모집글 ID (FK)
  user_id varchar(100) [not null]                   //작성자 ID (FK)
  content text [not null]                           //댓글 내용
  created_at timestamp [not null, default: `now()`] //작성일시
}

//recruitments 관계
Ref fk_recruitments_user_id: recruitments.user_id > users.id
Ref fk_recruitments_theme_id: recruitments.theme_id >? themes.id

//recruitment_applications 관계
Ref fk_recruitment_applications_recruitment_id: recruitment_applications.recruitment_id > recruitments.id [delete: cascade]
Ref fk_recruitment_applications_user_id: recruitment_applications.user_id > users.id

Ref fk_party_comments_recruitment_id: party_comments.recruitment_id > recruitments.id [delete: cascade]
Ref fk_party_comments_user_id: party_comments.user_id > users.id

Table bookmarks { //북마크

Ref fk_bookmarks_theme_id: bookmarks.theme_id > themes.id [delete: cascade]
